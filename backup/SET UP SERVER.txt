python3 - <<'PY'
from pathlib import Path

encodings = ["utf-8-sig", "cp1258", "windows-1252", "latin-1"]

for p in Path(".").rglob("*.py"):
    raw = p.read_bytes()

    try:
        raw.decode("utf-8")
        continue
    except UnicodeDecodeError:
        pass

    converted = False
    for enc in encodings:
        try:
            text = raw.decode(enc)
            p.write_text(text, encoding="utf-8", newline="\n")
            print(f"converted {p} from {enc} -> utf-8")
            converted = True
            break
        except Exception:
            pass

    if not converted:
        print(f"FAILED {p}")
PY
